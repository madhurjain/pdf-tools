# terms.txt: rules for redact_pdf.py, one per line.
#
#   old text => new text    remove "old text" and write "new text" in its place
#   old text                remove "old text" and cover the spot with a black box
#
# Lines starting with # and blank lines are ignored. Matching ignores case, and a
# dash or space in a term also matches any other dash, a space, or nothing.
#
# Every example below is commented out and uses made-up values. Remove the
# leading "# " from the lines you need and put in your own values.
# This file then holds your real personal data: don't commit or share it.

# --- Names ------------------------------------------------------------------
# List the full name first-to-last as it appears; longer terms are applied
# first, so the full name and a shorter form can both be listed.
# Jane Q Taxpayer => John Doe
# Jane Taxpayer => John Doe
# Taxpayer, Jane => Doe, John
# Some forms print names in capitals; the replacement is written as typed.
# JANE Q TAXPAYER => JOHN DOE

# --- Social Security number (SSN) / ITIN -------------------------------------
# Also matches 123–45–6789, 123 45 6789 and 123456789.
# 123-45-6789 => XXX-XX-XXXX
# A spouse's SSN or an ITIN works the same way.
# 900-70-1234 => XXX-XX-XXXX

# --- Employer identification number (EIN) ------------------------------------
# 12-3456789 => XX-XXXXXXX

# --- Address ----------------------------------------------------------------
# A term split across two lines in the PDF won't match, so give each line its
# own rule.
# 42 Elm Street => 1 Main Street
# 42 Elm St => 1 Main St
# Apt 3B => Apt 1
# Springfield, IL 62704 => Anytown, ST 00000

# --- Date of birth ----------------------------------------------------------
# 01/31/1980 => 01/01/1900

# --- Phone and email --------------------------------------------------------
# (555) 123-4567 => (555) 000-0000
# 555-123-4567 => 555-000-0000
# jane.taxpayer@example.com => name@example.com

# --- Bank accounts ----------------------------------------------------------
# No "=>": these are blacked out.
# 000123456789
# 021000021

# --- Employer, payer and other names ----------------------------------------
# Acme Payroll Services
# Acme Corporation => Example Employer Inc
